## 准备工作

- 操作系统：windows10或windows11
- 科学上网
- 磁盘空间预留50-60G

> 安装depot_tools时会自动安装python，如果机器曾经安装过python，建议完全删除，否则可能会导致安装失败



## 一、安装visual studio 2022

- 自行在网上下载vs2022，社区版即可

- 勾选C++桌面开发，通用windows平台开发

- 必须勾选的组件

  - Windows 10 SDK(10.0.1904) 和 Windows 11 SDK(10.0.2262)
  - C++ ATL for v142生成工具（x86和x64）
  - C++ MFC for v142生成工具（x86和x64）

- 如果vs2022安装过，可以打开vs安装器，点击修改后选择单个组件安装

- 安装完成后，将vs的安装路径添加到系统环境变量PATH中

  `D:\Program Files (x86)\Microsoft Visual Studio\2019\Community`

## 二、安装并配置win10 SDK

- 在控制面板 > 程序 > 程序和功能中找到windows SDK 10.0.2262
- 右键选择更改
- 选择change
- 找到Debugging Tools For Windows并勾选，点击change

## 三、配置git代理及系统代理

- 搭建梯子并确认代理端口（例如7890）

  ```shell
  #1. 设置git代理，在git bash中执行
  git config --global http.proxy 127.0.0.1:7890
  git config --global https.proxy 127.0.0.1:7890
  git config --global core.gitproxy 127.0.0.1:7890
  
  #2. 设置系统代理，cmd中执行
  set http_proxy=http://127.0.0.1:7890
  set https_proxy=http://127.0.0.1:7890
  ```

- 如果cmd窗口关闭，需要重新设置系统代理

- 编译完成后，记得取消git代理

## 四、安装depot_tools

- 下载depot_tools

  `git clone https://chromium.googlesource.com/chromium/tools/depot_tools.git`

- 将depot_tools的路径放到系统环境变量PATH中（一定要放在第一行）

  `D:\tools\depot_tools`

- 运行如下命令

  ```shell
  cd D:\tools\depot_tools
  gclient
  #在第一次运行时，gclient 将安装编译所需的所有windows 特定工具，包括 msysgit 和 python等
  ```

- 添加系统环境变量 DEPOT_TOOLS_WIN_TOOLCHAIN=0

## 五、获取webrtc源码

- 点击链接（https://chromiumdash.appspot.com/branches）确认版本是否支持vs版本（具体路径：webrtc\checkout\src\build\vs_toolchain.py）

  ```shell
  #示例
  MSVS_VERSIONS = collections.OrderedDict([
      ('2022', '17.0'),  # Default and packaged version of Visual Studio.
      ('2019', '16.0'),
      ('2017', '15.0'),
  ])
  ```

- 按照以下步骤进行

  ```shell
  mkdir webrtc
  cd webrtc
  fetch --nohooks webrtc   #下载.gclient的源码下载配置文件， gclient是webrtc的源码管理工具
  ```

- 配置gsutil代理

  - 创建 `C:\Users\<你的用户名>\.boto`文件（用户名不得为中文）

  - 文件内容如下:

    ```shell
    [Boto]
    proxy = 代理ip
    proxy_port = 代理端口
    ```

  - 设置环境变量:`set NO_AUTH_BOTO_CONFIG=C:\Users\<你的用户名>\.boto`

- 执行`gclient sync`  进行同步。如果出现网络中断或其他问题，使用`ctrl+c`后再次执行gclient sync

- 切换至webrtc m130/6723版本

  ```shell
  # 切换分支
  git checkout -b m130/6723 branch-heads/6723
  # 重新同步
  gclient sync
  ```

  

## 六、编译安装

- 按照如下步骤设置环境变量

  ```shell
  set GYP_MSVS_VERSION=2022
  set DEPOT_TOOLS_UPDATE=0
  set DEPOT_TOOLS_WIN_TOOLCHAIN=0
  set vs2022_install=D:\Program Files\Microsoft Visual Studio\2022\Community #vs安装路径
  set GYP_MSVS_OVERRIDE_PATH=D:\Program Files\Microsoft Visual Studio\2022\Community #vs安装路径
  set GYP_GENERATORS=msvs-ninja,ninja
  set WINDOWSSDKDIR=D:\Windows Kits\10
  ```

- 生成webrtc

  ```shell
  #进入webrtc源码路径
  cd D:\tools\webrtc\src
  
  #执行gn命令生成构建文件
  # 使用use_custom_libcxx=false解决链接问题
  gn gen --ide=vs2022 out/Release --args="is_debug=false target_cpu=\"x64\" use_custom_libcxx=false use_custom_libcxx_for_host=true"
  
  #支持h264编译
  gn gen --ide=vs2022 out/release_h264 --args="is_debug=false target_cpu=\"x64\" use_custom_libcxx=false use_custom_libcxx_for_host=true rtc_enable_protobuf=true ffmpeg_branding=\"Chrome\" rtc_use_h264=true proprietary_codecs=true symbol_level=0"
  
  ## gn相关参数
  # -ide：生成vs工程文件
  # -args：配置参数
  # 配置参数如下：
      # 为true编译Debug版本，false编译Release版本
  	# is_debug
  	
  	# 是否使用动态运行期库，设置false表示使用静态运行期库，Release版本将对应MT，Debug版将对应MTd
  	# is_component_build
  	
  	# 是否使用protobuf，使用可将其设置为true
  	# rtc_enable_protobuf
  	
  	# 禁用 webrtc自带的libc++库，防止和标准库冲突，如果编译动态库可以打开
      # use_custom_libcxx=false
      
      # 使用非 webrtc自带的libc++库，防止和标准库冲突
      # use_custom_libcxx_for_host=true 
      
      # 编译出的WebRTC库是否带符号表，这个数据量很大，会影响运行速度，所以一般设置为0，表示编译出的WebRTC不带符号表
  	# symbol_level
  	
  	#编译WebRTC时是否编译测试用例，如果为false则不编译，这样可以大大加快WebRTC的编译速度
  	# rtc_include_tests
  	
  	#启用lld
  	# use_lld
  	
  	#启用clang编译
  	# is_clang
  
  #执行ninja命令进行编译
  ninja -C  out/Release  -j 8
  ```



## 七、代码环境搭建

> 2024年10月29日更新，对于示例peerconnection client无法编译问题作出调整

- 打开vs installer，选择更改->单个组件，搜索clang，安装相关组件

- 编译器选择x64-Clang-Release，并设置配置类型为Release（默认为RelWithDegInfo），工具集选择clang_cl_x64_x64

- cmake进行如下设置

  ```cmake
  # 1. c++版本必须大于等于17
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  
  
  # 2. 由于webrtc编译的release版本需要链接静态线程库，所以设置MT。如果是debug版本根据编译提示进行修改，可能是MD或MDd
  set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -MT")
  set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -MT")
  
  # 3. 链接相关头文件
  set(LIBWEBRTC_PATH "D:/tools/webrtc2/src")
  list(APPEND PROJECT_INCLUDE ${LIBWEBRTC_PATH})
  list(APPEND PROJECT_INCLUDE "${LIBWEBRTC_PATH}/third_party/abseil-cpp")
  list(APPEND PROJECT_INCLUDE "${LIBWEBRTC_PATH}/third_party/jsoncpp/source/include")
  list(APPEND PROJECT_INCLUDE "${LIBWEBRTC_PATH}/third_party/libyuv/include")
  
  # 4. 链接相关库文件及中间文件
  # 此处GLOB_RECURSE表示深度搜索指定文件路径下所有.lib/.obj文件
  # lib文件：直接搜索编译webrtc生成的文件路径；obj文件：根据编译错误提示 unresolved symbol 查找所需的obj文件（主要用到test和第三方库中的json、absl）
  file(GLOB_RECURSE WEBRTC_LIB "${LIBWEBRTC_PATH}/out/Release/obj/*.lib")
  file(GLOB_RECURSE WEBRTC_OBJ "D:/tools/webrtc2/webrtc/lib/*.obj")
  list(APPEND PROJECT_LIBS ${WEBRTC_OBJ} ${WEBRTC_LIB})
  
  # 5. 链接相关系统库
  list(APPEND PROJECT_LIBS 
  advapi32.lib 
  comdlg32.lib 
  dbghelp.lib 
  dnsapi.lib 
  gdi32.lib 
  msimg32.lib 
  odbc32.lib 
  odbccp32.lib 
  oleaut32.lib 
  shell32.lib 
  shlwapi.lib 
  user32.lib 
  usp10.lib 
  uuid.lib 
  version.lib 
  wininet.lib 
  winmm.lib 
  winspool.lib 
  ws2_32.lib 
  delayimp.lib 
  kernel32.lib 
  ole32.lib 
  crypt32.lib 
  iphlpapi.lib 
  secur32.lib 
  dmoguids.lib 
  wmcodecdspuuid.lib 
  amstrmid.lib 
  msdmo.lib 
  strmiids.lib)
  
  # 6. 设置预编译内容
  add_definitions(
  -DUSE_AURA=1 
  -D_HAS_NODISCARD 
  -D_CRT_NONSTDC_NO_WARNINGS 
  -D_WINSOCK_DEPRECATED_NO_WARNINGS 
  "-DCR_CLANG_REVISION=\"llvmorg-20-init-3847-g69c43468-28\"" 
  -D_LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_NONE 
  -D_HAS_EXCEPTIONS=0 
  -D__STD_C 
  -D_CRT_RAND_S 
  -D_CRT_SECURE_NO_DEPRECATE 
  -D_SCL_SECURE_NO_DEPRECATE 
  -D_ATL_NO_OPENGL 
  -D_WINDOWS 
  -DCERT_CHAIN_PARA_HAS_EXTRA_FIELDS 
  -DPSAPI_VERSION=2 
  -DWIN32 
  -D_SECURE_ATL 
  -DWINAPI_FAMILY=WINAPI_FAMILY_DESKTOP_APP 
  -DWIN32_LEAN_AND_MEAN 
  -DNOMINMAX 
  -D_UNICODE 
  -DUNICODE 
  -DNTDDI_VERSION=NTDDI_WIN10_NI 
  -D_WIN32_WINNT=0x0A00 
  -DWINVER=0x0A00 
  -DNDEBUG 
  -DNVALGRIND 
  -DDYNAMIC_ANNOTATIONS_ENABLED=0 
  -DWEBRTC_ENABLE_PROTOBUF=1 
  -DWEBRTC_STRICT_FIELD_TRIALS=0 
  -DWEBRTC_INCLUDE_INTERNAL_AUDIO_DEVICE 
  -DWEBRTC_USE_H264 //设置使用h264
  -DRTC_ENABLE_VP9 
  -DRTC_DAV1D_IN_INTERNAL_DECODER_FACTORY 
  -DWEBRTC_HAVE_SCTP 
  -DWEBRTC_ENABLE_AVX2
  ```
